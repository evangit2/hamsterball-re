# Level Names and Colors Config File
# Format1: LevelName= #RRGGBB  (hex RGB, like HTML colors)
# Format2: LevelName= "string" (text string between " ")
# Lines starting with ; or # are comments
#
# PREFIXES:
# (Under the "LEVELS" section)
#   LevelName=	     #RRGGBB  — applies to BOTH timer and menu
#   board:LevelName= #RRGGBB  — only timer oval/text during gameplay
#   menu:LevelName=  #RRGGBB  — only race selection menu text
#   name:LevelName=  (string) — name of the Race.
#
# (Under the "ARENAS" section)
#   ArenaName=	     #RRGGBB  — applies to BOTH timer and menu
#   board:ArenaName= #RRGGBB  — only timer oval/text during gameplay
#   menu:ArenaName=  #RRGGBB  — only arena selection menu text
#   name:ArenaName=  (string) — name of the Arena.
#
# Re-reads every 2 seconds. Edit at runtime!
